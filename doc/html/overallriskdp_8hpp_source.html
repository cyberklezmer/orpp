59 class taylorreward : public Reward
60 {
61 public:
62 taylorreward(const finitepolicy& base,
63 const std::vector<double>& gradient,
64 const Reward& orig) : Reward(orig),
65 fbase(base), fgradient(gradient)
66 {
67 assert(fbase.size() == fgradient.size());
68 }
69 double operator() (const dpcondition<unsigned int,
70 typename ConstrainedActionSpace::Element_t>& x) const
71 {
72 assert(x.s < fgradient.size());
73 assert(x.s < fbase.size());
74 return std::max(0.0,Reward::operator()(x)+(x.a-fbase[x.s]) * fgradient[x.s]);
75 }
76 double maxgradadd(const ConstrainedActionSpace& space) const
77 {
78 double m = 0;
79 for(unsigned i=0; i<fgradient.size(); i++)
80 {
81 double x = 0;
82 if(fgradient[i] < 0)
83 {
84 typename ConstrainedActionSpace::Element_t a;
85 bool exists = space.firstfeasible(a,i);
86 assert(exists);
87 if(!exists)
88 throw exception("maxgradadd: At least one feasible element needed");
89 assert(a <= fbase[i]);
90 x = - (fbase[i]-a) * fgradient[i];
91 }
92 else if(fgradient[i] > 0)
93 {
94 typename ConstrainedActionSpace::Element_t a;
95 bool exists = space.lastfeasible(a,i);
96 assert(exists);
97 if(!exists)
98 throw exception("maxgradadd: At least one feasible element needed");
99 assert(a >= fbase[i]);
100 x = (fbase[i]-a) * fgradient[i];
101 }
102 m = std::max(m,x);
103 }
104 return m;
105 }

107 private:
108 finitepolicy fbase;
109 std::vector<double> fgradient;

232 double findiota(const nestedproblem& problem,
233 const finitepolicy& policy,
234 double constant,
235 const finitevaluefunction initvalue,
236 orpp::index initindex,
237 double accuracy,
238 const typename nestedproblem::computationparams& pars
239 ) const
240 {
241 rmtermination term(accuracy / 2);
242
243 rmdifference<nestedproblem> dif(problem,policy,constant,initvalue, initindex ,term, pars);
244
245 using boost::math::tools::bisect;
246 double from = 0; // The solution must lie in the interval [from, to], additionally f(from) <= 0 && f(to) >= 0
247 double to = this->fcrit.getparam();
248 assert(from <= to);
249 double dfrom = dif(from);
250 double dto = dif(to);
251 if(fabs(dfrom)<accuracy)
252 return from;
253 else if(fabs(dto)<accuracy)
254 return to;
255 else if(from < to)
256 {
257 if(dfrom < 0)
258 throw exception("Error finding nested r.a. parameter: dif(from) < 0");
259 else if(dto > 0)
260 throw exception("Error finding nested r.a. parameter: dif(to) > 0");
261 else
262 {
263 try
264 {
265 std::pair<double, double> result =
266 bisect(dif, from, to, term);
267 return (result.first + result.second) / 2;
268 }
269 catch(...)
270 {
271 sys::logline() << std::endl << "error finding dif(iota)=0, constant=" << constant << std::endl;
272 const unsigned nsteps = 5;
273 for(unsigned i=0; i<=nsteps; i++)
274 {
275 double x = from+ (to-from)/nsteps * i;
276 sys::log() << "dif(" << x << ")=" << dif(x) << std::endl;
277 }
278 throw;
279 }
280 }
281 }
282 else
283 throw exception("Error finding nested r.a. parameter: from > to");
284 }
